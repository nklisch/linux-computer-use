cmake_minimum_required(VERSION 3.22)
project(LcuMonitor VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
include(CTest)
include(GNUInstallDirs)
# The Rust contract owns protocol revision; do not maintain a second constant.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../src/desktop/mod.rs")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../src/desktop/mod.rs" wire_line REGEX "pub const WIRE_REVISION:")
string(REGEX MATCH "= ([0-9]+)" wire_match "${wire_line}")
set(wire_revision "${CMAKE_MATCH_1}")
add_library(monitor_core STATIC
  src/daemon_client.cpp src/monitor_model.cpp src/monitor_window.cpp src/preview_widget.cpp)
target_include_directories(monitor_core PUBLIC src)
target_compile_definitions(monitor_core PRIVATE LCU_WIRE_REVISION=${wire_revision})
target_link_libraries(monitor_core PUBLIC Qt6::Widgets Qt6::Network)
target_compile_options(monitor_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lcu-monitor src/main.cpp)
target_link_libraries(lcu-monitor PRIVATE monitor_core)
install(TARGETS lcu-monitor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
configure_file(packaging/org.nklisch.LcuMonitor.desktop.in org.nklisch.LcuMonitor.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.nklisch.LcuMonitor.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(monitor_test tests/monitor_test.cpp)
  target_link_libraries(monitor_test PRIVATE monitor_core Qt6::Test)
  add_test(NAME monitor COMMAND monitor_test)
  set_tests_properties(monitor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()
